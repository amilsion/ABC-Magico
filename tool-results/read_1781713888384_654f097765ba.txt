  1476→        function playTone(freq, duration, type = 'sine') {
  1477→            const osc = audioCtx.createOscillator();
  1478→            const gain = audioCtx.createGain();
  1479→            osc.connect(gain);
  1480→            gain.connect(audioCtx.destination);

  1483→            gain.gain.setValueAtTime(0.3, audioCtx.currentTime);
  1484→            gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + duration);
  1485→            osc.start(audioCtx.currentTime);
  1486→            osc.stop(audioCtx.currentTime + duration);
  1487→        }
